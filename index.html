/* ============================================================ الرفيق — sw.js Service Worker الإصدار: 2.0.0 الهدف: - تحديث آمن للكاش - تحميل style.css مع index.html - دعم العمل بدون إنترنت - التخلص من كاش الإصدارات القديمة ============================================================ */ 'use strict'; /* ============================================================ 1) إعدادات الكاش ============================================================ */ const CACHE_VERSION = 'rafeeq-v2'; const APP_CACHE = `${CACHE_VERSION}-app`; const DATA_CACHE = `${CACHE_VERSION}-data`; /* ============================================================ 2) ملفات التطبيق الأساسية ============================================================ */ const APP_FILES = [ './', './index.html', './style.css', './manifest.json', './app.js', './theme.js', './prayer.html', './quran.html', './adhkar.html', './hisnul.html', './arbaeen.html', './tafsir.html', './qibla.html', './more.html', './icon-192.png', './icon-512.png', './adhan.mp3' ]; /* ============================================================ 3) ملفات البيانات ============================================================ */ const DATA_FILES = [ './quran-local.json', './tafsir-saadi.json' ]; /* ============================================================ 4) تثبيت Service Worker الجديد ============================================================ */ self.addEventListener('install', event => { console.log( '📦 الرفيق: تثبيت Service Worker الجديد', CACHE_VERSION ); event.waitUntil( Promise.all([ caches.open(APP_CACHE) .then(cache => { console.log( '📁 تخزين ملفات التطبيق...' ); return cache.addAll(APP_FILES); }), caches.open(DATA_CACHE) .then(cache => { console.log( '📚 تخزين ملفات البيانات...' ); return cache.addAll(DATA_FILES); }) ]) .then(() => { console.log( '✅ تم تخزين ملفات الرفيق بنجاح' ); /* * تفعيل النسخة الجديدة فورًا. * هذا مهم حتى لا يبقى الإصدار القديم مسيطرًا. */ return self.skipWaiting(); }) .catch(error => { console.error( '❌ فشل تثبيت Service Worker:', error ); throw error; }) ); }); /* ============================================================ 5) تفعيل Service Worker ============================================================ */ self.addEventListener('activate', event => { console.log( '🔄 تفعيل Service Worker:', CACHE_VERSION ); event.waitUntil( caches.keys() .then(cacheNames => { return Promise.all( cacheNames .filter(cacheName => { return ( cacheName.startsWith('rafeeq-') && cacheName !== APP_CACHE && cacheName !== DATA_CACHE ); }) .map(oldCache => { console.log( '🗑️ حذف الكاش القديم:', oldCache ); return caches.delete( oldCache ); }) ); }) .then(() => { console.log( '✅ تم تنظيف الكاش القديم' ); /* * جعل النسخة الجديدة تتحكم في الصفحات المفتوحة. */ return self.clients.claim(); }) ); }); /* ============================================================ 6) التعامل مع طلبات الملفات ============================================================ */ self.addEventListener('fetch', event => { const request = event.request; /* * نتعامل فقط مع GET */ if (request.method !== 'GET') { return; } event.respondWith( caches.match(request) .then(cachedResponse => { /* * إذا كان الملف موجودًا في الكاش * نستخدمه مباشرة. */ if (cachedResponse) { return cachedResponse; } /* * إذا لم يكن موجودًا نحاول تحميله من الإنترنت. */ return fetch(request) .then(networkResponse => { /* * التحقق من الاستجابة. */ if ( !networkResponse || networkResponse.status !== 200 || networkResponse.type === 'opaque' ) { return networkResponse; } /* * نسخ الاستجابة قبل تخزينها. */ const responseClone = networkResponse.clone(); /* * تحديد نوع الكاش. */ const requestURL = new URL(request.url); const isDataFile = requestURL.pathname.endsWith('.json'); const cacheName = isDataFile ? DATA_CACHE : APP_CACHE; /* * حفظ الملف في الكاش. */ caches.open(cacheName) .then(cache => { cache.put( request, responseClone ); }); return networkResponse; }) .catch(error => { console.warn( '⚠️ فشل الاتصال:', request.url ); /* * إذا كانت صفحة HTML * نعيد index.html من الكاش. */ if ( request.destination === 'document' ) { return caches.match( './index.html' ); } throw error; }); }) ); }); /* ============================================================ 7) استقبال الأوامر من التطبيق ============================================================ */ self.addEventListener('message', event => { if (!event.data) { return; } /* * طلب تفعيل النسخة الجديدة. */ if ( event.data.type === 'SKIP_WAITING' ) { self.skipWaiting(); } /* * طلب تنظيف الكاش يدويًا. */ if ( event.data.type === 'CLEAR_CACHE' ) { event.waitUntil( caches.keys() .then(cacheNames => { return Promise.all( cacheNames .filter(name => name.startsWith('rafeeq-') ) .map(name => caches.delete(name) ) ); }) ); } }); /* ============================================================ 8) رسالة جاهزية ============================================================ */ console.log( '🚀 الرفيق — Service Worker جاهز', CACHE_VERSION );
